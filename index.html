async function deleteAppointment(id) { if (!confirm("¿Deseas eliminar esta cita permanentemente? Se borrará de la pantalla y de la hoja de Google Sheets.")) { return; } // 1. Eliminación visual y en almacenamiento local inmediata const prev = getAppointments(); const updated = prev.filter(a => String(a.id).trim() !== String(id).trim()); saveAppointments(updated); renderAdminTable(); renderDays(); renderSlots(); // 2. Notificación a Google Sheets const url = getScriptUrl(); if (url) { try { const endpoint = `${url}${url.includes('?') ? '&' : '?'}action=delete&id=${encodeURIComponent(id)}&_t=${Date.now()}`; // Se envía la orden a Google Apps Script (probando POST y GET como respaldo) try { await fetch(endpoint, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ action: "delete", id: id }) }); } catch (postErr) { // Si el navegador bloquea el POST por redirección CORS, GET lo ejecuta sin preflight await fetch(endpoint, { method: "GET" }); } } catch (e) { console.error("Error al sincronizar borrado con Google Sheets:", e); } // Breve espera para que Google Sheets procese y confirme la eliminación antes de consultar setTimeout(() => { refreshFromCloud(); }, 800); } } async function clearAllAppointments() { if (!confirm("¿Estás segura de que deseas borrar TODA la lista de citas? Esta acción limpiará tanto la pantalla como el Google Sheet y reiniciará el turno al primer asesor.")) { return; } // 1. Limpieza inmediata local saveAppointments([]); renderAdminTable(); renderDays(); renderSlots(); // 2. Limpieza en Google Sheets const url = getScriptUrl(); if (url) { try { const endpoint = `${url}${url.includes('?') ? '&' : '?'}action=clearAll&_t=${Date.now()}`; try { await fetch(endpoint, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ action: "clearAll" }) }); } catch (postErr) { await fetch(endpoint, { method: "GET" }); } } catch (e) { console.error("Error al limpiar citas en Google Sheets:", e); } setTimeout(() => { refreshFromCloud(); }, 800); } alert("La lista de citas ha sido limpiada por completo."); }
